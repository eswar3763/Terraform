1. Get the application URL:
  export POD_NAME=$(kubectl get pods --namespace {{ .Release.Namespace }} -l "app.kubernetes.io/name={{ include "order-service.name" . }},app.kubernetes.io/instance={{ .Release.Name }}" -o jsonpath="{.items[0].metadata.name}")
  kubectl --namespace {{ .Release.Namespace }} port-forward $POD_NAME 8080:8080

2. Check deployment status:
  kubectl rollout status deployment/{{ include "order-service.fullname" . }} -n {{ .Release.Namespace }}

3. View application logs:
  kubectl logs -n {{ .Release.Namespace }} -l app.kubernetes.io/name={{ include "order-service.name" . }} -f

4. For more information, visit:
  https://github.com/eswar3763/Terraform
